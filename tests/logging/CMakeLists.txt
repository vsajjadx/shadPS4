cmake_minimum_required(VERSION 3.23)
project(shadps4_runtime_logging_regressions LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
get_filename_component(SHAD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPDLOG_INSTALL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(fmt SOURCE_DIR "${SHAD_ROOT}/externals/fmt" OVERRIDE_FIND_PACKAGE)
FetchContent_MakeAvailable(fmt)
add_subdirectory("${SHAD_ROOT}/externals/spdlog" "${CMAKE_CURRENT_BINARY_DIR}/spdlog" EXCLUDE_FROM_ALL)
add_executable(runtime_logging_test runtime_logging_test.cpp)
target_compile_features(runtime_logging_test PRIVATE cxx_std_20)
target_include_directories(runtime_logging_test PRIVATE "${SHAD_ROOT}/src")
target_link_libraries(runtime_logging_test PRIVATE spdlog::spdlog)
add_test(NAME runtime_logging_regressions COMMAND runtime_logging_test)
set_tests_properties(runtime_logging_regressions PROPERTIES TIMEOUT 30)

add_executable(runtime_correctness_test runtime_correctness_test.cpp
    "${SHAD_ROOT}/src/common/hack_features.cpp")
target_compile_features(runtime_correctness_test PRIVATE cxx_std_20)
target_include_directories(runtime_correctness_test PRIVATE "${SHAD_ROOT}/src")
target_link_libraries(runtime_correctness_test PRIVATE fmt::fmt Threads::Threads)
add_test(NAME verbose_budget_window COMMAND runtime_correctness_test budget)
add_test(NAME verbose_budget_concurrent COMMAND runtime_correctness_test concurrent)
add_test(NAME verbose_macro_short_circuit COMMAND runtime_correctness_test macro)
add_test(NAME title_nhl_serial COMMAND runtime_correctness_test title CUSA37933 "NHL 24" 0 1)
add_test(NAME title_nhl_regional COMMAND runtime_correctness_test title UNKNOWN "NHL® 24" 0 1)
add_test(NAME title_other_game COMMAND runtime_correctness_test title CUSA57220 "Another title" 0 0)
add_test(NAME title_unknown COMMAND runtime_correctness_test title UNKNOWN "" 0 0)
add_test(NAME title_order_serial COMMAND runtime_correctness_test title CUSA00035 "" 1 0)
add_test(NAME title_order_regional COMMAND runtime_correctness_test title UNKNOWN "The Order: 1886" 1 0)

add_test(NAME runtime_verbose_policy COMMAND runtime_correctness_test runtime-policy)
add_test(NAME build_completion_publication COMMAND runtime_correctness_test completion)
set_tests_properties(build_completion_publication PROPERTIES TIMEOUT 10)
